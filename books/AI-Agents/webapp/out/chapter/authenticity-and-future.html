15.1 Agent Identity

The Identity Problem

When a human sends an email, you can verify who they are. When an AI agent sends an email on behalf of a human, a company, or another agent -- who is actually responsible?

IDENTITY CHAIN:

Human: "Book me a flight to Paris"
  |
  v
Agent A (Personal Assistant)
  |  "I will find flights and book the best one"
  v
Agent B (Travel Booking Service)
  |  "Searching 50 airlines for best price"
  v
Agent C (Airline Booking System)
  |  "Booked flight AF123 for $450"
  v
Confirmation sent to human

Question: If flight AF123 is cancelled, who is responsible?
- Agent A? (it chose Agent B)
- Agent B? (it searched airlines)
- Agent C? (it made the booking)
- The human? (they asked for it)
- The airline? (they cancelled)

Why Agent Identity Matters

ScenarioWithout IdentityWith Identity
Agent sends emailRecipient cannot verify senderCryptographic proof of agent + human
Agent makes purchaseNo audit trail of who authorizedSigned authorization chain
Agent signs contractLegal ambiguityVerified agent + human delegation
Two agents negotiateNeither can prove identityMutual authentication
Agent produces contentCannot verify originWatermarked + signed output

Building Agent Identity

"""Agent Identity System - Cryptographic identity for AI agents."""

import hashlib
import json
import time
from dataclasses import dataclass, field
from cryptography.hazmat.primitives import hashes, serialization
from cryptography.hazmat.primitives.asymmetric import rsa, padding


@dataclass
class AgentIdentity:
    """Cryptographic identity for an AI agent."""
    agent_id: str
    name: str
    owner: str  # Human or organization that controls this agent
    created_at: float = field(default_factory=time.time)
    capabilities: list[str] = field(default_factory=list)
    public_key: str = ""
    private_key: str = ""

    def generate_keys(self):
        """Generate RSA key pair for signing."""
        private_key = rsa.generate_private_key(
            public_exponent=65537,
            key_size=2048,
        )
        self.private_key = private_key.private_bytes(
            encoding=serialization.Encoding.PEM,
            format=serialization.PrivateFormat.PKCS8,
            encryption_algorithm=serialization.NoEncryption(),
        ).decode()
        self.public_key = private_key.public_key().public_bytes(
            encoding=serialization.Encoding.PEM,
            format=serialization.PublicFormat.SubjectPublicKeyInfo,
        ).decode()

    def sign(self, message: str) -> str:
        """Sign a message with the agent's private key."""
        private_key = serialization.load_pem_private_key(
            self.private_key.encode(), password=None
        )
        signature = private_key.sign(
            message.encode(),
            padding.PKCS1v15(),
            hashes.SHA256(),
        )
        return signature.hex()

    def to_certificate(self) -> dict:
        """Generate an agent certificate (public info only)."""
        return {
            "agent_id": self.agent_id,
            "name": self.name,
            "owner": self.owner,
            "created_at": self.created_at,
            "capabilities": self.capabilities,
            "public_key": self.public_key,
            "fingerprint": hashlib.sha256(
                self.public_key.encode()
            ).hexdigest()[:16],
        }


@dataclass
class SignedAction:
    """A cryptographically signed action by an agent."""
    agent_id: str
    action_type: str
    action_data: dict
    timestamp: float = field(default_factory=time.time)
    human_authorization: str | None = None
    signature: str = ""

    def to_signable_string(self) -> str:
        return json.dumps({
            "agent_id": self.agent_id,
            "action_type": self.action_type,
            "action_data": self.action_data,
            "timestamp": self.timestamp,
            "human_authorization": self.human_authorization,
        }, sort_keys=True)


# Usage
agent = AgentIdentity(
    agent_id="agent_001",
    name="TravelBot",
    owner="John Smith",
    capabilities=["book_flights", "search_hotels", "check_weather"],
)
agent.generate_keys()

action = SignedAction(
    agent_id="agent_001",
    action_type="book_flight",
    action_data={"flight": "AF123", "date": "2025-03-15", "passenger": "John Smith"},
    human_authorization="auth_token_abc123",
)
action.signature = agent.sign(action.to_signable_string())

cert = agent.to_certificate()
print(f"Agent Fingerprint: {cert['fingerprint']}")

15.2 Watermarking AI Content

Text Watermarking

"""Text watermarking for AI-generated content."""

import hashlib
import random


class TextWatermark:
    """Embed and detect watermarks in AI-generated text."""

    def __init__(self, secret_key: str):
        self.secret_key = secret_key

    def _get_rng(self, text_seed: str) -> random.Random:
        """Create a deterministic PRNG from the text + secret key."""
        seed = hashlib.sha256(
            f"{self.secret_key}:{text_seed}".encode()
        ).hexdigest()
        return random.Random(seed)

    def embed(self, text: str, agent_id: str) -> str:
        """Embed a watermark into text using zero-width characters."""
        words = text.split()
        if len(words) < 10:
            return text

        rng = self._get_rng(agent_id)
        num_modifications = max(1, len(words) // 20)
        indices = sorted(
            rng.sample(range(len(words)), min(num_modifications, len(words)))
        )

        watermarked = words.copy()
        for idx in indices:
            if rng.random() > 0.5:
                watermarked[idx] = watermarked[idx] + "​"
            else:
                watermarked[idx] = watermarked[idx] + "‌"

        return " ".join(watermarked)

    def detect(self, text: str, agent_id: str) -> dict:
        """Detect if text contains a watermark."""
        rng = self._get_rng(agent_id)
        words = text.split()
        matches = 0
        total = 0

        num_modifications = max(1, len(words) // 20)
        indices = sorted(
            rng.sample(range(len(words)), min(num_modifications, len(words)))
        )

        for idx in indices:
            if idx < len(words):
                total += 1
                if "​" in words[idx] or "‌" in words[idx]:
                    matches += 1

        confidence = matches / total if total > 0 else 0
        return {
            "is_watermarked": confidence > 0.5,
            "confidence": confidence,
            "matches": matches,
            "checked": total,
        }


# Usage
watermarker = TextWatermark(secret_key="my-secret-key-2025")
original = "This is an AI-generated response about machine learning."
watermarked = watermarker.embed(original, agent_id="agent_001")

result = watermarker.detect(watermarked, agent_id="agent_001")
print(f"Watermarked: {result['is_watermarked']}")   # True
print(f"Confidence: {result['confidence']:.2f}")     # 1.00

wrong = watermarker.detect(watermarked, agent_id="agent_999")
print(f"Wrong agent: {wrong['is_watermarked']}")     # False

C2PA Content Provenance

"""C2PA-style content provenance metadata."""

import json
import hashlib
from datetime import datetime, timezone


class ContentProvenance:
    """Create and verify content provenance manifests (C2PA-inspired)."""

    def create_manifest(
        self,
        content: bytes,
        agent_id: str,
        agent_name: str,
        human_creator: str | None = None,
    ) -> dict:
        """Create a provenance manifest for content."""
        content_hash = hashlib.sha256(content).hexdigest()

        manifest = {
            "claim_generator": agent_name,
            "agent_id": agent_id,
            "human_creator": human_creator,
            "content_hash": f"sha256:{content_hash}",
            "created_at": datetime.now(timezone.utc).isoformat(),
            "software": {
                "agent": agent_name,
                "model": "gpt-4o",
                "version": "1.0.0",
            },
            "assertions": [
                {"label": "c2pa.created", "data": datetime.now(timezone.utc).isoformat()},
                {"label": "c2pa.ai_generated", "data": True},
                {"label": "c2pa.ai_tool", "data": agent_name},
            ],
        }

        manifest_bytes = json.dumps(manifest, sort_keys=True).encode()
        manifest["signature"] = hashlib.sha256(manifest_bytes).hexdigest()
        return manifest

    def verify_manifest(self, manifest: dict, content: bytes) -> dict:
        """Verify a content provenance manifest."""
        expected = hashlib.sha256(content).hexdigest()
        actual = manifest.get("content_hash", "").replace("sha256:", "")

        return {
            "valid": expected == actual,
            "ai_generated": manifest.get("assertions", [{}])[1].get("data", False),
            "agent": manifest.get("claim_generator", "unknown"),
            "human_creator": manifest.get("human_creator"),
        }

Building Trust Through Transparency

"""Transparency features for trustworthy AI agents."""


class TransparencyLayer:
    """Add transparency to agent responses."""

    def explain_decision(self, decision: dict) -> str:
        """Generate a human-readable explanation of an agent decision."""
        parts = []
        parts.append("## How I arrived at this answer:\n")

        if decision.get("sources"):
            parts.append("### Sources used:")
            for i, source in enumerate(decision["sources"], 1):
                parts.append(
                    f"{i}. {source['name']} (relevance: {source['score']:.0%})"
                )
            parts.append("")

        if decision.get("reasoning_steps"):
            parts.append("### My reasoning:")
            for step in decision["reasoning_steps"]:
                parts.append(f"- {step}")
            parts.append("")

        confidence = decision.get("confidence", 0)
        parts.append(f"### Confidence: {confidence:.0%}")
        if confidence < 0.5:
            parts.append(
                "*Note: I am not very confident. Please verify with a human.*"
            )
        elif confidence < 0.8:
            parts.append("*Note: This answer may need additional verification.*")

        if decision.get("alternatives"):
            parts.append("\n### Other possibilities considered:")
            for alt in decision["alternatives"]:
                parts.append(f"- {alt}")

        return "\n".join(parts)

    def generate_confidence_report(
        self, query: str, answer: str, sources: list[dict]
    ) -> dict:
        """Generate a confidence report for a response."""
        scores = {
            "source_quality": 0,
            "answer_completeness": 0,
            "source_agreement": 0,
        }

        if sources:
            scores["source_quality"] = sum(
                s.get("score", 0) for s in sources
            ) / len(sources)

        word_count = len(answer.split())
        if word_count > 50:
            scores["answer_completeness"] = 0.9
        elif word_count > 20:
            scores["answer_completeness"] = 0.7
        else:
            scores["answer_completeness"] = 0.4

        scores["source_agreement"] = min(len(sources) / 3, 1.0)
        overall = sum(scores.values()) / len(scores)

        return {
            "overall_confidence": overall,
            "breakdown": scores,
            "recommendation": (
                "high" if overall > 0.75
                else "medium" if overall > 0.5
                else "low"
            ),
        }

15.4 Autonomous Agents

The Autonomy Spectrum

LEVEL 0: Tool
  Human does everything. Tool helps.
  Example: Calculator

LEVEL 1: Copilot
  Human leads. AI assists.
  Example: GitHub Copilot

LEVEL 2: Agent
  AI executes tasks with human oversight.
  Example: Customer support agent with escalation

LEVEL 3: Autonomous Agent
  AI plans and executes independently. Human sets goals.
  Example: AI research assistant

LEVEL 4: Multi-Agent System
  Multiple AI agents collaborate without human involvement.
  Example: AI development team (planner + coder + tester)

LEVEL 5: Self-Improving Agent
  AI modifies its own code and improves over time.
  Example: AutoGPT-style systems (experimental)

Building an Autonomous Research Agent

"""Autonomous research agent - plans and executes research tasks."""

import asyncio
from dataclasses import dataclass, field


@dataclass
class ResearchTask:
    topic: str
    questions: list[str] = field(default_factory=list)
    findings: list[dict] = field(default_factory=list)
    status: str = "pending"


class AutonomousResearcher:
    """Agent that independently researches topics."""

    def __init__(self, llm, search_tool, web_scraper):
        self.llm = llm
        self.search = search_tool
        self.scraper = web_scraper

    async def research(self, topic: str, depth: int = 2) -> dict:
        """Autonomously research a topic."""
        questions = await self._plan_research(topic)

        findings = []
        for question in questions[:depth * 3]:
            answer = await self._research_question(question)
            findings.append({
                "question": question,
                "answer": answer,
                "sources": answer.get("sources", []),
            })

        synthesis = await self._synthesize(topic, findings)
        report = await self._generate_report(topic, synthesis, findings)

        return {
            "topic": topic,
            "questions_researched": len(findings),
            "findings": findings,
            "synthesis": synthesis,
            "report": report,
        }

    async def _plan_research(self, topic: str) -> list[str]:
        """Use LLM to generate research questions."""
        response = await self.llm.generate(
            f"Generate 5 focused research questions about: {topic}"
        )
        return response  # Simplified

    async def _research_question(self, question: str) -> dict:
        """Research a single question."""
        search_results = await self.search.search(question)
        contents = []
        for result in search_results[:3]:
            content = await self.scraper.scrape(result["url"])
            contents.append(content)

        context = "\n\n".join(contents[:3])
        answer = await self.llm.generate(
            f"Based on this information:\n{context}\n\n"
            f"Answer this question: {question}"
        )
        return {
            "answer": answer,
            "sources": [r["url"] for r in search_results[:3]],
        }

    async def _synthesize(self, topic: str, findings: list[dict]) -> str:
        """Synthesize all findings into a coherent narrative."""
        findings_text = "\n".join(
            f"- {f['question']}: {f['answer'][:200]}"
            for f in findings
        )
        return await self.llm.generate(
            f"Synthesize these research findings about {topic}:\n{findings_text}"
        )

    async def _generate_report(self, topic: str, synthesis: str,
                                findings: list[dict]) -> str:
        """Generate a final research report."""
        return await self.llm.generate(
            f"Write a comprehensive research report on {topic}.\n"
            f"Synthesis: {synthesis}\n"
            f"Key findings: {len(findings)} questions answered."
        )

Safety Constraints for Autonomous Agents

"""Safety constraints for autonomous agents."""


class AutonomousAgentGuardrails:
    """Safety constraints that limit agent autonomy."""

    def __init__(self, config: dict):
        self.max_cost_per_task = config.get("max_cost_per_task", 10.0)
        self.max_steps = config.get("max_steps", 20)
        self.max_time_seconds = config.get("max_time_seconds", 300)
        self.blocked_actions = config.get("blocked_actions", [
            "delete_file", "send_email", "make_payment",
        ])
        self.require_approval = config.get("require_approval", [
            "purchase", "publish", "share_data", "contact_human",
        ])

    def check_action(self, action: dict, state: dict) -> dict:
        """Check if an action is allowed."""
        reasons = []

        if state.get("steps_taken", 0) >= self.max_steps:
            reasons.append(f"Max steps ({self.max_steps}) reached")

        if state.get("total_cost", 0) >= self.max_cost_per_task:
            reasons.append(f"Max cost (${self.max_cost_per_task}) reached")

        if state.get("elapsed_seconds", 0) >= self.max_time_seconds:
            reasons.append(f"Max time ({self.max_time_seconds}s) reached")

        if action.get("type") in self.blocked_actions:
            reasons.append(f"Action '{action['type']}' is blocked")

        needs_approval = action.get("type") in self.require_approval

        return {
            "allowed": len(reasons) == 0 and not needs_approval,
            "needs_approval": needs_approval,
            "reasons": reasons,
        }

Agent Marketplace API

"""Agent Marketplace - Buy and sell agent capabilities."""

from fastapi import FastAPI, HTTPException
from pydantic import BaseModel
import time

app = FastAPI(title="Agent Marketplace")


class AgentListing(BaseModel):
    name: str
    description: str
    capabilities: list[str]
    pricing_model: str  # "per_execution", "per_hour", "subscription"
    price_per_unit: float
    rating: float = 0.0
    total_executions: int = 0


class ExecutionRequest(BaseModel):
    input_data: dict
    max_cost: float = 1.0
    timeout_seconds: int = 60


class ExecutionResponse(BaseModel):
    output_data: dict
    cost: float
    execution_time_ms: float
    agent_id: str


# In-memory store
agent_listings: dict[str, AgentListing] = {}


@app.post("/agents")
async def list_agent(listing: AgentListing):
    """List a new agent on the marketplace."""
    agent_id = f"agent_{len(agent_listings) + 1:04d}"
    agent_listings[agent_id] = listing
    return {"agent_id": agent_id, "status": "listed"}


@app.get("/agents")
async def search_agents(capability: str = None, max_price: float = None):
    """Search for agents by capability and price."""
    results = []
    for agent_id, listing in agent_listings.items():
        if capability and capability not in listing.capabilities:
            continue
        if max_price and listing.price_per_unit > max_price:
            continue
        results.append({
            "agent_id": agent_id,
            "name": listing.name,
            "price": listing.price_per_unit,
            "capabilities": listing.capabilities,
        })
    return results


@app.post("/agents/{agent_id}/execute")
async def execute_agent(agent_id: str, request: ExecutionRequest):
    """Execute an agent and pay for the result."""
    if agent_id not in agent_listings:
        raise HTTPException(404, "Agent not found")

    listing = agent_listings[agent_id]
    start_time = time.time()

    estimated_cost = listing.price_per_unit
    if estimated_cost > request.max_cost:
        raise HTTPException(400, "Cost exceeds budget")

    # Execute (placeholder)
    result = {"result": "executed", "input_received": True}
    execution_time = (time.time() - start_time) * 1000
    listing.total_executions += 1

    return ExecutionResponse(
        output_data=result,
        cost=estimated_cost,
        execution_time_ms=execution_time,
        agent_id=agent_id,
    )

The Human-Agent Collaboration Model

"""Human-Agent Collaboration Framework."""


class CollaborationLevel:
    ASSISTED = "assisted"             # Agent suggests, human decides
    COLLABORATIVE = "collaborative"   # Agent and human work together
    SUPERVISED = "supervised"         # Agent acts, human reviews
    AUTONOMOUS = "autonomous"         # Agent acts independently


class HumanAgentTeam:
    """A team of humans and agents working together."""

    def __init__(self):
        self.humans: list[dict] = []
        self.agents: list[dict] = []
        self.tasks: list[dict] = []

    def assign_task(self, task: dict) -> dict:
        """Assign a task to the best available worker."""
        complexity = task.get("complexity", "medium")
        risk = task.get("risk_level", "low")
        requires_creativity = task.get("creativity_needed", False)

        if risk == "high" or complexity == "expert":
            return {
                "assigned_to": "human",
                "reason": "High risk requires human judgment",
                "agent_assist": True,
            }
        elif requires_creativity and complexity != "simple":
            return {
                "assigned_to": "human_agent_team",
                "reason": "Creative task benefits from both",
                "mode": "collaborative",
            }
        elif complexity == "simple":
            return {
                "assigned_to": "agent",
                "reason": "Routine task, agent handles autonomously",
                "human_review": risk != "low",
            }
        else:
            return {
                "assigned_to": "agent",
                "reason": "Medium complexity, agent with human review",
                "mode": "supervised",
            }

    def get_utilization(self) -> dict:
        """Get utilization metrics for the team."""
        return {
            "humans": {
                "total": len(self.humans),

Implementing Ethical Constraints

"""Ethical framework implementation for AI agents."""

from enum import Enum


class EthicalPrinciple(Enum):
    TRANSPARENCY = "transparency"
    FAIRNESS = "fairness"
    PRIVACY = "privacy"
    SAFETY = "safety"
    ACCOUNTABILITY = "accountability"
    HUMAN_OVERSIGHT = "human_oversight"


class EthicalConstraints:
    """Enforce ethical constraints on agent behavior."""

    def __init__(self):
        self.principles = {
            EthicalPrinciple.TRANSPARENCY: {
                "requires_explanation": True,
                "must_cite_sources": True,
            },
            EthicalPrinciple.FAIRNESS: {
                "check_for_bias": True,
                "demographic_parity": True,
            },
            EthicalPrinciple.PRIVACY: {
                "pii_handling": "redact",
                "consent_required": True,
                "data_retention_days": 90,
            },
            EthicalPrinciple.SAFETY: {
                "content_filter": True,
                "escalation_required": True,
            },
            EthicalPrinciple.ACCOUNTABILITY: {
                "log_all_actions": True,
                "human_audit_trail": True,
            },
            EthicalPrinciple.HUMAN_OVERSIGHT: {
                "max_autonomous_actions": 5,
                "require_approval_for": [
                    "financial_transactions",
                    "personal_data_access",
                    "external_communications",
                ],
            },
        }

    def evaluate_action(self, action: dict) -> dict:
        """Evaluate an action against ethical constraints."""
        violations = []

        # Check transparency
        if action.get("is_factual_claim") and not action.get("sources"):
            violations.append({
                "principle": "transparency",
                "message": "Factual claims must include sources",
            })

        # Check privacy
        if action.get("contains_pii"):
            violations.append({
                "principle": "privacy",
                "message": "Action contains PII - consent required",
            })

        # Check human oversight
        approval_required = self.principles[
            EthicalPrinciple.HUMAN_OVERSIGHT
        ]["require_approval_for"]
        if action.get("type") in approval_required:
            if not action.get("human_approved"):
                violations.append({
                    "principle": "human_oversight",
                    "message": f"Action '{action['type']}' requires human approval",
                })

        return {
            "ethical": len(violations) == 0,
            "violations": violations,
        }

Bias Detection and Mitigation

"""Bias detection and mitigation for AI agents."""

import re


class BiasDetector:
    """Detect bias in agent outputs."""

    def __init__(self):
        self.demographic_terms = {
            "gender": ["he", "she", "man", "woman", "male", "female"],
            "race": ["white", "black", "asian", "hispanic", "latino"],
            "age": ["young", "old", "elderly", "teenager", "senior"],
        }

    def analyze_text(self, text: str) -> dict:
        """Analyze text for potential bias."""
        text_lower = text.lower()
        findings = []

        stereotyping_patterns = [
            (r"(men|women)\s+are\s+(all|always|never)", "gender_stereotype"),
            (r"(black|white|asian)\s+people\s+(always|never)", "racial_stereotype"),
            (r"(old|young)\s+people\s+(cannot|shouldn't)", "age_stereotype"),
        ]

        for pattern, bias_type in stereotyping_patterns:
            if re.search(pattern, text_lower):
                findings.append({
                    "type": bias_type,
                    "severity": "high",
                })

        for category, terms in self.demographic_terms.items():
            mentions = sum(text_lower.count(term) for term in terms)
            if mentions > 0:
                findings.append({
                    "type": "representation",
                    "category": category,
                    "mentions": mentions,
                })

        return {
            "bias_detected": len(findings) > 0,
            "findings": findings,
            "severity": (
                "high" if any(f["severity"] == "high" for f in findings)
                else "medium" if findings
                else "none"
            ),
        }

    def mitigate_bias(self, prompt: str) -> str:
        """Add bias mitigation instructions to prompts."""
        mitigation = """

IMPORTANT BIAS MITIGATION GUIDELINES:
- Use gender-neutral language when gender is not relevant
- Avoid stereotyping based on race, gender, age, or religion
- Represent diverse perspectives when discussing groups
- Use person-first language
- When providing examples, use diverse names and demographics
"""
        return prompt + mitigation
