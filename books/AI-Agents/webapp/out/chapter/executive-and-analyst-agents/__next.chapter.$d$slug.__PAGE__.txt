## 5.1 What Executive Agents Do

Executive agents are AI systems designed to help business leaders make better decisions faster. They analyze data, generate reports, summarize meetings, and provide actionable insights — all automatically.

### The Problem They Solve

Executives spend hours each week on:
- Reading through lengthy reports
- Analyzing spreadsheets
- Summarizing meeting notes
- Drafting emails and presentations
- Tracking KPIs across departments

An executive agent automates these tasks, freeing up time for strategic thinking.

### Key Capabilities

| Capability | What It Does | Time Saved |
|-----------|-------------|-----------|
| **Report Generation** | Turns raw data into executive summaries | 2-4 hours/week |
| **Data Analysis** | Queries databases and generates insights | 3-5 hours/week |
| **KPI Monitoring** | Tracks metrics and alerts on anomalies | 1-2 hours/week |
| **Meeting Summarization** | Extracts action items from recordings | 2-3 hours/week |
| **Email Drafting** | Writes professional communications | 1-2 hours/week |

## 5.2 Report Generation

### From Data to Narrative

The key challenge isn't generating text — it's understanding what the data means and communicating it clearly.

```python
"""
Executive Report Generator Agent
Turns raw data into clear, actionable executive summaries.
"""

from openai import OpenAI
from dataclasses import dataclass
from datetime import datetime
import json


@dataclass
class ReportSection:
    title: str
    data: dict
    narrative: str


class ExecutiveReportAgent:
    """Generates executive reports from raw business data."""
    
    def __init__(self, api_key: str | None = None):
        self.client = OpenAI(api_key=api_key)
        self.model = "gpt-4o"
    
    def analyze_data(self, data: dict) -> str:
        """Analyze raw data and generate executive narrative."""
        
        prompt = f"""You are a senior business analyst writing for a CEO.
        
Analyze this data and write a clear, concise executive summary.
Focus on:
1. Key findings (what does the data show?)
2. Trends (what's changing?)
3. Risks (what could go wrong?)
4. Recommendations (what should we do?)

Data:
{json.dumps(data, indent=2, default=str)}

Write in a professional but accessible tone. Use bullet points for clarity.
Keep it under 500 words."""
        
        response = self.client.chat.completions.create(
            model=self.model,
            messages=[
                {"role": "system", "content": "You are an expert business analyst."},
                {"role": "user", "content": prompt}
            ],
            temperature=0.3
        )
        
        return response.choices[0].message.content
    
    def generate_full_report(
        self, 
        company_name: str,
        period: str,
        sections: list[dict]
    ) -> str:
        """Generate a complete executive report."""
        
        report_parts = [
            f"# Executive Report: {company_name}",
            f"**Period:** {period}",
            f"**Generated:** {datetime.now().strftime('%Y-%m-%d %H:%M')}",
            "",
            "---",
            ""
        ]
        
        for section in sections:
            title = section["title"]
            data = section["data"]
            
            # Analyze each section
            narrative = self.analyze_data(data)
            
            report_parts.extend([
                f"## {title}",
                "",
                narrative,
                "",
                "---",
                ""
            ])
        
        # Generate overall summary
        all_data = {s["title"]: s["data"] for s in sections}
        summary = self.generate_overall_summary(all_data)
        
        report_parts.extend([
            "## Overall Assessment",
            "",
            summary,
            "",
            "---",
            f"*Report generated by Executive Report Agent*"
        ])
        
        return "\n".join(report_parts)
    
    def generate_overall_summary(self, all_data: dict) -> str:
        """Generate an overall business summary."""
        
        prompt = f"""Based on these business metrics, provide a 2-paragraph 
executive overview covering overall business health, key priorities, 
and strategic recommendations.

{json.dumps(all_data, indent=2, default=str)}"""
        
        response = self.client.chat.completions.create(
            model=self.model,
            messages=[
                {"role": "user", "content": prompt}
            ],
            temperature=0.3
        )
        
        return response.choices[0].message.content


# === Usage ===
agent = ExecutiveReportAgent()

# Sample data
sections = [
    {
        "title": "Revenue Performance",
        "data": {
            "total_revenue": 2450000,
            "growth_rate": 0.12,
            "by_product": {
                "Enterprise": 1500000,
                "SMB": 650000,
                "Self-serve": 300000
            },
            "churn_rate": 0.05
        }
    },
    {
        "title": "Customer Metrics",
        "data": {
            "total_customers": 1250,
            "new_customers": 180,
            "nps_score": 72,
            "support_tickets": 340,
            "avg_resolution_time_hours": 4.2
        }
    }
]

report = agent.generate_full_report(
    company_name="Acme Corp",
    period="Q4 2025",
    sections=sections
)

print(report)

## 5.3 Data Analysis Pipelines

### SQL Agent for Business Queries

```python
"""
SQL Data Analysis Agent
Natural language to SQL, with automatic visualization.
"""

import sqlite3
from openai import OpenAI
from dataclasses import dataclass
import json


@dataclass
class AnalysisResult:
    question: str
    sql_query: str
    results: list[dict]
    insight: str


class DataAnalysisAgent:
    """Converts natural language questions to SQL and analyzes results."""
    
    def __init__(self, db_path: str, api_key: str | None = None):
        self.db_path = db_path
        self.client = OpenAI(api_key=api_key)
        self.conn = sqlite3.connect(db_path)
        self.conn.row_factory = sqlite3.Row
    
    def get_schema(self) -> str:
        """Get database schema for context."""
        cursor = self.conn.execute(
            "SELECT name FROM sqlite_master WHERE type='table'"
        )
        tables = [row[0] for row in cursor.fetchall()]
        
        schema_parts = []
        for table in tables:
            cursor = self.conn.execute(f"PRAGMA table_info({table})")
            columns = [(row[1], row[2]) for row in cursor.fetchall()]
            col_str = ", ".join(f"{name} ({dtype})" for name, dtype in columns)
            schema_parts.append(f"Table: {table}\nColumns: {col_str}")
        
        return "\n\n".join(schema_parts)
    
    def nl_to_sql(self, question: str) -> str:
        """Convert natural language question to SQL."""
        
        schema = self.get_schema()
        
        prompt = f"""Convert this natural language question to a SQL query.

Database Schema:
{schema}

Question: {question}

Return ONLY the SQL query, no explanation. Use SQLite syntax."""
        
        response = self.client.chat.completions.create(
            model="gpt-4o",
            messages=[
                {"role": "user", "content": prompt}
            ],
            temperature=0.0
        )
        
        sql = response.choices[0].message.content.strip()
        # Clean up markdown code blocks
        if sql.startswith("```"):
            sql = sql.split("\n", 1)[1].rsplit("```", 1)[0]
        
        return sql
    
    def execute_query(self, sql: str) -> list[dict]:
        """Execute SQL query and return results."""
        try:
            cursor = self.conn.execute(sql)
            columns = [desc[0] for desc in cursor.description]
            rows = cursor.fetchall()
            return [dict(zip(columns, row)) for row in rows]
        except Exception as e:
            return [{"error": str(e)}]
    
    def analyze(self, question: str) -> AnalysisResult:
        """Full analysis pipeline: question → SQL → results → insight."""
        
        # Generate SQL
        sql = self.nl_to_sql(question)
        
        # Execute
        results = self.execute_query(sql)
        
        # Generate insight
        prompt = f"""Given this business question and SQL results, provide 
a clear, actionable insight in 2-3 sentences.

Question: {question}
SQL: {sql}
Results: {json.dumps(results, default=str)}

Focus on what the data means for business decisions."""
        
        response = self.client.chat.completions.create(
            model="gpt-4o",
            messages=[{"role": "user", "content": prompt}],
            temperature=0.3
        )
        
        return AnalysisResult(
            question=question,
            sql_query=sql,
            results=results,
            insight=response.choices[0].message.content
        )


# === Usage ===
agent = DataAnalysisAgent("business_data.db")
result = agent.analyze("What were our top 5 customers by revenue last quarter?")
print(f"SQL: {result.sql_query}")
print(f"Insight: {result.insight}")
```

## 5.4 KPI Monitoring

### Automated Metric Tracking

```python
"""
KPI Monitoring Agent
Tracks key metrics and alerts when thresholds are breached.
"""

from dataclasses import dataclass
from datetime import datetime
from enum import Enum


class AlertSeverity(Enum):
    INFO = "info"
    WARNING = "warning"
    CRITICAL = "critical"


@dataclass
class KPI:
    name: str
    current_value: float
    target: float
    threshold_warning: float  # % below target to trigger warning
    threshold_critical: float  # % below target to trigger critical
    trend: str  # "up", "down", "stable"


@dataclass
class KPIAlert:
    kpi: KPI
    severity: AlertSeverity
    message: str
    recommendation: str


class KPIMonitoringAgent:
    """Monitors KPIs and generates alerts with recommendations."""
    
    def __init__(self):
        self.kpis: dict[str, KPI] = {}
        self.history: list[dict] = []
    
    def register_kpi(self, kpi: KPI):
        """Register a KPI to monitor."""
        self.kpis[kpi.name] = kpi
    
    def check_kpi(self, kpi: KPI) -> KPIAlert | None:
        """Check if a KPI needs attention."""
        
        if kpi.current_value >= kpi.target:
            return None  # KPI is on target
        
        deviation = (kpi.target - kpi.current_value) / kpi.target
        
        if deviation >= kpi.threshold_critical:
            severity = AlertSeverity.CRITICAL
            message = f"CRITICAL: {kpi.name} is {deviation:.1%} below target"
        elif deviation >= kpi.threshold_warning:
            severity = AlertSeverity.WARNING
            message = f"WARNING: {kpi.name} is {deviation:.1%} below target"
        else:
            return None
        
        # Generate recommendation based on KPI type
        recommendation = self._generate_recommendation(kpi, severity)
        
        return KPIAlert(
            kpi=kpi,
            severity=severity,
            message=message,
            recommendation=recommendation
        )
    
    def _generate_recommendation(self, kpi: KPI, severity: AlertSeverity) -> str:
        """Generate actionable recommendation for a KPI alert."""
        
        recommendations = {
            "revenue": "Review sales pipeline and marketing spend. Consider promotional campaigns.",
            "churn": "Analyze churn reasons. Implement retention campaigns for at-risk customers.",
            "nps": "Survey recent detractors. Address top complaints immediately.",
            "conversion": "Review funnel drop-off points. A/B test landing pages.",
            "support_time": "Add support staff or implement AI triage for faster resolution.",
        }
        
        for key, rec in recommendations.items():
            if key in kpi.name.lower():
                return rec
        
        return f"Investigate root cause for {kpi.name} and develop action plan."
    
    def run_check(self) -> list[KPIAlert]:
        """Check all registered KPIs and return alerts."""
        alerts = []
        
        for kpi in self.kpis.values():
            alert = self.check_kpi(kpi)
            if alert:
                alerts.append(alert)
        
        # Sort by severity
        severity_order = {
            AlertSeverity.CRITICAL: 0,
            AlertSeverity.WARNING: 1,
            AlertSeverity.INFO: 2
        }
        alerts.sort(key=lambda a: severity_order[a.severity])
        
        return alerts


# === Usage ===
monitor = KPIMonitoringAgent()

# Register KPIs
monitor.register_kpi(KPI(
    name="monthly_revenue",
    current_value=180000,
    target=200000,
    threshold_warning=0.05,
    threshold_critical=0.15,
    trend="down"
))

monitor.register_kpi(KPI(
    name="customer_churn_rate",
    current_value=0.08,
    target=0.05,
    threshold_warning=0.2,
    threshold_critical=0.5,
    trend="up"
))

# Check all KPIs
alerts = monitor.run_check()
for alert in alerts:
    print(f"\n{alert.severity.value.upper()}: {alert.message}")
    print(f"Recommendation: {alert.recommendation}")
```

## 5.5 Meeting Summarization

### Extracting Action Items Automatically

```python
"""
Meeting Summarization Agent
Processes meeting transcripts and extracts key information.
"""

from openai import OpenAI
from dataclasses import dataclass


@dataclass
class MeetingSummary:
    topic: str
    summary: str
    decisions: list[str]
    action_items: list[dict]  # [{"task": ..., "owner": ..., "deadline": ...}]
    open_questions: list[str]


class MeetingSummaryAgent:
    """Summarizes meetings and extracts action items."""
    
    def __init__(self, api_key: str | None = None):
        self.client = OpenAI(api_key=api_key)
    
    def summarize(self, transcript: str) -> MeetingSummary:
        """Process a meeting transcript and extract key information."""
        
        prompt = f"""Analyze this meeting transcript and extract:

1. **Topic**: One-line summary of what the meeting was about
2. **Summary**: 3-5 sentence executive summary
3. **Decisions Made**: List of concrete decisions
4. **Action Items**: Tasks with assigned owners and deadlines
5. **Open Questions**: Unresolved items needing follow-up

Transcript:
{transcript}

Return as JSON with these exact keys:
topic, summary, decisions, action_items, open_questions"""
        
        response = self.client.chat.completions.create(
            model="gpt-4o",
            messages=[
                {"role": "system", "content": "You are a meeting analyst."},
                {"role": "user", "content": prompt}
            ],
            temperature=0.2,
            response_format={"type": "json_object"}
        )
        
        import json
        data = json.loads(response.choices[0].message.content)
        
        return MeetingSummary(
            topic=data.get("topic", ""),
            summary=data.get("summary", ""),
            decisions=data.get("decisions", []),
            action_items=data.get("action_items", []),
            open_questions=data.get("open_questions", [])
        )
    
    def format_summary(self, summary: MeetingSummary) -> str:
        """Format summary as a clean report."""
        
        lines = [
            f"# Meeting Summary: {summary.topic}",
            "",
            "## Summary",
            summary.summary,
            "",
            "## Decisions Made",
        ]
        
        for decision in summary.decisions:
            lines.append(f"- {decision}")
        
        lines.extend(["", "## Action Items"])
        
        for item in summary.action_items:
            task = item.get("task", "")
            owner = item.get("owner", "Unassigned")
            deadline = item.get("deadline", "No deadline")
            lines.append(f"- [ ] {task} — **{owner}** (due: {deadline})")
        
        if summary.open_questions:
            lines.extend(["", "## Open Questions"])
            for q in summary.open_questions:
                lines.append(f"- {q}")
        
        return "\n".join(lines)


# === Usage ===
agent = MeetingSummaryAgent()

transcript = """
John: Let's review the Q4 numbers. Revenue is up 12% but churn increased.
Sarah: The enterprise segment is strong. SMB is where we're losing people.
Mike: I think we need to improve onboarding. New users drop off after day 3.
John: Good point. Mike, can you audit the onboarding flow?
Sarah: I'll put together a churn analysis report by Friday.
John: Decisions: We're investing in onboarding improvements. Budget approved for Q1.
Mike: I'll have the audit done by next Wednesday.
"""

summary = agent.summarize(transcript)
print(agent.format_summary(summary))
```

"""
Email Drafting Agent
Generates professional emails based on context and intent.
"""

from openai import OpenAI


class EmailDraftingAgent:
    """Drafts professional emails based on context."""
    
    def __init__(self, api_key: str | None = None):
        self.client = OpenAI(api_key=api_key)
    
    def draft_email(
        self,
        recipient: str,
        context: str,
        intent: str,
        tone: str = "professional",
        length: str = "medium"
    ) -> str:
        """Draft an email based on context and intent."""
        
        length_guide = {
            "short": "3-4 sentences",
            "medium": "1 paragraph, 5-7 sentences",
            "long": "2-3 paragraphs"
        }
        
        prompt = f"""Draft a {tone} email.

To: {recipient}
Context: {context}
Intent: {intent}
Length: {length_guide.get(length, 'medium')}

Write only the email body. No subject line. No greeting like 'Dear X' 
(use a natural opening). No signature block."""
        
        response = self.client.chat.completions.create(
            model="gpt-4o",
            messages=[
                {"role": "user", "content": prompt}
            ],
            temperature=0.7
        )
        
        return response.choices[0].message.content
    
    def reply_to_email(
        self,
        original_email: str,
        desired_response: str,
        tone: str = "professional"
    ) -> str:
        """Draft a reply to an existing email."""
        
        prompt = f"""Draft a reply to this email.

Original email:
{original_email}

Desired response intent: {desired_response}
Tone: {tone}

Write only the reply body."""
        
        response = self.client.chat.completions.create(
            model="gpt-4o",
            messages=[{"role": "user", "content": prompt}],
            temperature=0.7
        )
        
        return response.choices[0].message.content


# === Usage ===
agent = EmailDraftingAgent()

email = agent.draft_email(
    recipient="engineering team",
    context="We need to migrate from the old database to PostgreSQL by end of Q1",
    intent="Inform about migration timeline and ask for volunteer leads",
    tone="professional but friendly"
)

print(email)

"""
Decision Support Agent
Helps executives evaluate options and make informed decisions.
"""

from openai import OpenAI
from dataclasses import dataclass


@dataclass
class DecisionOption:
    name: str
    pros: list[str]
    cons: list[str]
    cost: str
    timeline: str
    risk_level: str


class DecisionSupportAgent:
    """Analyzes decisions and provides structured recommendations."""
    
    def __init__(self, api_key: str | None = None):
        self.client = OpenAI(api_key=api_key)
    
    def analyze_options(
        self,
        decision: str,
        options: list[str],
        constraints: list[str]
    ) -> str:
        """Analyze multiple options for a decision."""
        
        prompt = f"""Analyze this business decision.

Decision: {decision}

Options:
{chr(10).join(f'{i+1}. {opt}' for i, opt in enumerate(options))}

Constraints:
{chr(10).join(f'- {c}' for c in constraints)}

For each option, provide:
- Pros (2-3)
- Cons (2-3)
- Estimated cost
- Timeline
- Risk level (low/medium/high)

Then give your recommendation with reasoning.

Format as structured markdown."""
        
        response = self.client.chat.completions.create(
            model="gpt-4o",
            messages=[
                {"role": "system", "content": "You are a strategic business advisor."},
                {"role": "user", "content": prompt}
            ],
            temperature=0.3
        )
        
        return response.choices[0].message.content


# === Usage ===
agent = DecisionSupportAgent()

analysis = agent.analyze_options(
    decision="Choose a cloud provider for our AI infrastructure",
    options=[
        "AWS SageMaker",
        "Google Vertex AI",
        "Azure ML",
        "Self-hosted with vLLM"
    ],
    constraints=[
        "Budget: $50k/month max",
        "Must support GPU instances",
        "Need SOC2 compliance",
        "Team has AWS experience"
    ]
)

print(analysis)
